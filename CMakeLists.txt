cmake_minimum_required(VERSION 3.20)
project(log_ingest VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
option(LOG_INGEST_BENCHMARKS "Build Google Benchmark workloads" OFF)
set(LOG_INGEST_SANITIZER "" CACHE STRING "Sanitizer: address or thread")
if(LOG_INGEST_SANITIZER AND NOT LOG_INGEST_SANITIZER MATCHES "^(address|thread)$")
  message(FATAL_ERROR "Unsupported sanitizer")
endif()
add_library(log_ingest INTERFACE)
target_include_directories(log_ingest INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(log_ingest INTERFACE cxx_std_20)
target_link_libraries(log_ingest INTERFACE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(log_ingest INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(LOG_INGEST_SANITIZER)
    target_compile_options(log_ingest INTERFACE -fsanitize=${LOG_INGEST_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(log_ingest INTERFACE -fsanitize=${LOG_INGEST_SANITIZER})
    if(LOG_INGEST_SANITIZER STREQUAL "address")
      target_compile_options(log_ingest INTERFACE -fsanitize=undefined)
      target_link_options(log_ingest INTERFACE -fsanitize=undefined)
    endif()
  endif()
endif()
add_executable(log-ingest src/main.cpp)
target_link_libraries(log-ingest PRIVATE log_ingest)
if(BUILD_TESTING)
  add_executable(log-ingest-tests tests/test_engine.cpp)
  target_link_libraries(log-ingest-tests PRIVATE log_ingest)
  add_test(NAME log-ingest-tests COMMAND log-ingest-tests)
  set_tests_properties(log-ingest-tests PROPERTIES TIMEOUT 60)
endif()
if(LOG_INGEST_BENCHMARKS)
  find_package(benchmark CONFIG REQUIRED)
  add_executable(queue-benchmark benchmarks/queue_benchmark.cpp)
  target_link_libraries(queue-benchmark PRIVATE log_ingest benchmark::benchmark_main)
endif()
